import { readFileSync } from 'node:fs';
import { deserializeWorld, serializeWorld } from './src/simulation/serialization';
import { simulationMinutesBetween, makeSimulationMoment } from './src/simulation/dates';
import { controlledCommitmentsBlockingMinuteAdvance, scheduledActivityState } from './src/simulation/time-work';
import { socialInvitationsFor, declineSocialInvitation } from './src/presentation/social-invitation';
const raw=JSON.parse(readFileSync('/private/tmp/ui-core-transfer-recovery/edu-saved-world.json','utf8'));
const world=deserializeWorld(serializeWorld(raw));
if(world.control.kind!=='person') throw new Error('Missing player');
const personId=world.control.personId;
const original=serializeWorld(world);
const invitation=socialInvitationsFor(world,personId).find(i=>i.activityId.endsWith('40f1b8f6333747bd'));
if(!invitation) throw new Error('Exact saved invitation missing');
if(serializeWorld(world)!==original) throw new Error('Read mutation');
const minutes=simulationMinutesBetween(world.currentMoment,{...world.currentMoment,date:'2026-01-14'});
const before=controlledCommitmentsBlockingMinuteAdvance(world,minutes);
const next=declineSocialInvitation(world,{personId,activityId:invitation.activityId,revision:invitation.revision});
const after=controlledCommitmentsBlockingMinuteAdvance(next,minutes);
if(!before.includes(invitation.activityId)||after.includes(invitation.activityId)) throw new Error('Hold disposition failed');
if(JSON.stringify(world.history.scheduledActivities)!==JSON.stringify(next.history.scheduledActivities)) throw new Error('Identity mutation');
if(JSON.stringify(world.currentMoment)!==JSON.stringify(next.currentMoment)) throw new Error('Clock moved');
if(serializeWorld(deserializeWorld(serializeWorld(next)))!==serializeWorld(next)) throw new Error('Reload mismatch');
console.log(JSON.stringify({currentMoment:world.currentMoment,invitation,before,after,status:scheduledActivityState(next,invitation.activityId).status,clockUnchanged:true,activitiesUnchanged:true,reload:true},null,2));
