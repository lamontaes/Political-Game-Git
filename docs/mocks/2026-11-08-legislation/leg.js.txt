/* clicks on the legislation screens; returns true when it handled one */
function legClick(e) {
  const t = (sel) => e.target.closest(sel);
  let el;
  if ((el = t("[data-llaw]"))) { hideTip(); $("#cardhost").innerHTML = lawCard(el.dataset.llaw); return true; }
  if ((el = t("[data-lamt]"))) { LB.amt = Math.max(300000, Math.min(2000000, LB.amt + 100000 * +el.dataset.lamt)); render(); return true; }
  if ((el = t("[data-lset]"))) { const [k, v] = el.dataset.lset.split(":"); LB[k] = k === "years" ? +v : v; render(); return true; }
  if ((el = t("[data-lpart]"))) { const k = el.dataset.lpart; legPart = k && legPart !== k ? k : null; render(); return true; }
  if (t("[data-lstaff]")) { legStaff = true; legView = "compare"; legPart = null; render(); return true; }
  if ((el = t("[data-lkeep]"))) { legStaff = false; legView = "clean"; render(); return true; }
  if ((el = t("[data-lview]"))) { const v = el.dataset.lview === "Compare" ? "compare" : "clean"; if (cur === "leg-track") trackView = v; else legView = v; render(); return true; }
  if ((el = t("[data-lidx]"))) { legIdx = el.dataset.lidx; legPart = null; render(); return true; }
  if ((el = t("[data-lvote]"))) { const [k, v] = el.dataset.lvote.split(":"); legVote[k] = legVote[k] === v ? null : v; render(); return true; }
  if ((el = t("[data-lpage]"))) { cur = "leg-" + el.dataset.lpage; legPart = null; closeCard(); render(); return true; }
  if (t(".lask")) return true;
  return false;
}
/* the compact panel sits just under the term it changes, inside the bill */
function placePop() {
  const pop = $(".legpop"), wrap = $(".lbillwrap"); if (!pop || !wrap) return;
  const tm = wrap.querySelector(`.lterm[data-lpart="${pop.dataset.for}"]`); if (!tm) return;
  const rects = tm.getClientRects(), last = rects[rects.length - 1], wr = wrap.getBoundingClientRect(), k = wr.width / wrap.offsetWidth;
  let top = (last.bottom - wr.top) / k + 8, left = (last.left - wr.left) / k - 12;
  left = Math.max(0, Math.min(left, wrap.offsetWidth - pop.offsetWidth));
  if (top + pop.offsetHeight > wrap.offsetHeight + 40) top = (rects[0].top - wr.top) / k - pop.offsetHeight - 8;
  pop.style.top = top + "px"; pop.style.left = left + "px";
}
