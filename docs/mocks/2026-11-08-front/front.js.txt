const frSet = (k, v) => {
  if (k === "stop") { frStop.has(v) ? frStop.delete(v) : frStop.add(v); return; }
  if (k === "mv") { const [id, val] = v.split("."); FR_MV[id] = val; FR.held = ""; return; }
  if (k === "margin" && FR.body === "Board of Aldermen") { const up = Number(v) > 0; const pool = up ? FR_BOARD.filter((i) => FR_MV[i] === "No") : FR_BOARD.filter((i) => FR_MV[i] === "Yes").reverse(); if (pool.length) FR_MV[pool[0]] = up ? "Yes" : "No"; FR.held = ""; return; }
  if (/^[+-]\d+$/.test(v) && typeof FR[k] === "number") FR[k] += Number(v);
  else FR[k] = /^\d+$/.test(v) && typeof FR[k] === "number" ? Number(v) : v.replace(/_/g, " ");
  if (k === "margin") FR.held = "";
  if (k === "body") { FR.margin = Math.floor(FR_SIZE[FR.body] / 2) + 1; FR.held = ""; FR.signed = ""; }
  if (k === "kind") { FR.sroleA = 0; FR.sroleB = 0; FR.played = ""; }
  if (k === "bribe" || k === "amt") FR.offered = "";
  if (k === "until" || k === "runN") FR.ran = "";
  if (k === "learned") FR.gloss = "";
};
