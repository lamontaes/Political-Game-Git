import { randomBytes } from "node:crypto";
import { mkdirSync, writeFileSync } from "node:fs";
import { execSync } from "node:child_process";
import { buildBenchWorld } from "/home/user/Political-Game-Git/src/bench/world";
import { gameStartDate } from "/home/user/Political-Game-Git/src/core2/game-start";
import { drawPlace } from "/home/user/Political-Game-Git/scripts/core2-year/place";
import { advanceCore } from "/home/user/Political-Game-Git/src/core2/life";
import { coreAPI } from "/home/user/Political-Game-Git/src/core2/state";
import { setBookkeeping } from "/home/user/Political-Game-Git/src/core2/bookkeeping";
import { addDays, makeIsoDate } from "/home/user/Political-Game-Git/src/simulation/dates";

const OUT = process.env.OUT!;
mkdirSync(OUT + "/tables", { recursive: true });
const seed = process.env.SEED ?? `${Date.now()}-${randomBytes(6).toString("hex")}`;
const clockByte = Date.now();
const span = Number(process.env.SPAN ?? 180 + (clockByte % 221));
setBookkeeping(false);
const place = drawPlace(seed, "stress-audit-laws");
const t0 = Date.now();
const world = buildBenchWorld(place.seed, gameStartDate(), { twoLevel: true, placeKey: place.key });
const core = world.core;
const start = core.date;
const end = addDays(makeIsoDate(start), span);
const head = execSync("git -C /home/user/Political-Game-Git rev-parse HEAD").toString().trim();
const meta = {
  seed, placeSeed: place.seed, placeKey: place.key, placeName: place.name, state: place.state,
  displayName: world.place.displayName, placeRecord: world.place, head, span, start, end,
  playerId: world.playerId, buildSeconds: (Date.now() - t0) / 1000,
};
console.log(JSON.stringify(meta));
writeFileSync(OUT + "/meta.json", JSON.stringify(meta, null, 1));
const replacer = (_k: string, v: unknown) =>
  v instanceof Map ? Object.fromEntries(v) : v instanceof Set ? [...v] : typeof v === "bigint" ? String(v) : v;
const dumpTables = (tag: string) => {
  const api = coreAPI(core);
  for (const m of core.modules.values()) {
    if (!m.state) continue;
    const store = api.moduleState(m.id);
    const tables: Record<string, unknown> = {};
    for (const name of Object.keys(m.state.tables)) tables[name] = store.table(name).rows();
    writeFileSync(`${OUT}/tables/${tag}-${m.id}.json`, JSON.stringify(tables, replacer));
  }
};
dumpTables("day0");
let day = start;
let n = 0;
const r0 = Date.now();
while (day < end) {
  const next = addDays(makeIsoDate(day), 1);
  advanceCore(core, next);
  day = next;
  if (++n % 30 === 0) console.log(`day ${n} ${day} ${((Date.now() - r0) / 1000).toFixed(0)}s`);
}
meta["runSeconds" as keyof typeof meta] = ((Date.now() - r0) / 1000) as never;
writeFileSync(OUT + "/meta.json", JSON.stringify(meta, null, 1));
dumpTables("end");
// Durable log
const log = [...core.durableLog.values()];
writeFileSync(OUT + "/durable-log.json", JSON.stringify(log, replacer));
const kinds = Object.fromEntries([...core.logByKind].map(([k, s]) => [k, s.size]));
writeFileSync(OUT + "/log-kinds.json", JSON.stringify(kinds, null, 1));
writeFileSync(OUT + "/journal.json", JSON.stringify({
  totals: core.cashJournal.totals,
  totalsByOwner: core.cashJournal.totalsByOwner,
  accounts: core.cashJournal.accounts,
  externalFlowsByOwner: core.cashJournal.externalFlowsByOwner,
}, replacer));
writeFileSync(OUT + "/public-orgs.json", JSON.stringify([...core.publicOrganizations.values()], replacer));
const player = core.people.get(world.playerId!);
writeFileSync(OUT + "/player.json", JSON.stringify(player, replacer));
writeFileSync(OUT + "/stopgap-hits.json", JSON.stringify([...core.stopgapHits]));
console.log("done", JSON.stringify(meta));

// ---- extra dumps for the audit ----
import { lawHere, lawInForce } from "/home/user/Political-Game-Git/src/core2/modules/laws";
import { inForceHere } from "/home/user/Political-Game-Git/src/core2/modules/law-effects";
{
  const api = coreAPI(core);
  const strip = (p: Record<string, unknown>) => {
    const { knowledge, log, ...rest } = p as Record<string, unknown>;
    return rest;
  };
  const leg = api.moduleState("core2-legislature-v1").table("votes").rows() as unknown as { memberId: string }[];
  const ids = new Set<string>(leg.map((v) => v.memberId));
  const player = core.people.get(world.playerId!)!;
  for (const p of core.people.values()) if (p.householdId === player.householdId) ids.add(p.id);
  const people: Record<string, unknown> = {};
  for (const id of ids) { const p = core.people.get(id); if (p) people[id] = strip(p as never); }
  writeFileSync(OUT + "/people.json", JSON.stringify(people, replacer));
  const rels = [...core.relationships.values()].filter((r) => ids.has((r as never as { fromId: string }).fromId) || ids.has((r as never as { personId: string }).personId));
  writeFileSync(OUT + "/relationships.json", JSON.stringify(rels, replacer));
  writeFileSync(OUT + "/organizations.json", JSON.stringify([...core.organizations.values()], replacer));
  writeFileSync(OUT + "/commitments.json", JSON.stringify([...core.work.commitments.values()].filter((c) => ids.has(c.personId)), replacer));
  const dates = ["2025-01-20", "2025-03-07", "2025-03-22", "2025-07-02", "2025-10-02", "2025-12-16", "2026-01-02", end];
  const law: Record<string, unknown> = {};
  for (const d of dates) {
    if (d > end) continue;
    law[d] = {
      here: lawHere(api, d),
      medicaidWork: lawInForce(api, "us-policy-positions:health-human-services.medicaid-work-requirement", d),
      immigration: lawInForce(api, "us-policy-positions:justice-public-safety.partner-with-federal-immigration-enforcement", d),
      groceries: lawInForce(api, "us-policy-positions:fiscal.exempt-groceries-from-sales-tax", d),
    };
  }
  law["inForceHereToday"] = {
    medicaidWork: inForceHere(api, "us-policy-positions:health-human-services.medicaid-work-requirement"),
    immigration: inForceHere(api, "us-policy-positions:justice-public-safety.partner-with-federal-immigration-enforcement"),
    preschool: inForceHere(api, "us-policy-positions:education.universal-preschool"),
  };
  writeFileSync(OUT + "/law-in-force-by-date.json", JSON.stringify(law, replacer));
  const pay = [...core.work.detailedPayroll.values()].filter((r) => ids.has(r.personId));
  writeFileSync(OUT + "/payroll-household.json", JSON.stringify(pay, replacer));
  console.log("extra dumps done");
}
